    %% Core Equipment Classes
    class ULyraEquipmentManagerComponent {
        -FLyraEquipmentList EquipmentList
        +EquipItem(EquipmentDefinition) ULyraEquipmentInstance
        +UnequipItem(ItemInstance)
        +GetFirstInstanceOfType() ULyraEquipmentInstance
    }
    
    class FLyraEquipmentList {
        <<struct>>
        -TArray~FLyraAppliedEquipmentEntry~ Entries
        +AddEntry(EquipmentDefinition) ULyraEquipmentInstance
        +RemoveEntry(Instance)
        +PostReplicatedAdd()
        +PreReplicatedRemove()
    }
    
    class FLyraAppliedEquipmentEntry {
        <<struct>>
        -EquipmentDefinition : TSubclassOf
        -Instance : ULyraEquipmentInstance*
        -GrantedHandles : FLyraAbilitySet_GrantedHandles
    }
    
    class ULyraEquipmentDefinition {
        <<DataAsset>>
        +InstanceType : TSubclassOf~ULyraEquipmentInstance~
        +AbilitySetsToGrant : TArray~ULyraAbilitySet~
        +ActorsToSpawn : TArray~FLyraEquipmentActorToSpawn~
    }
    
    class ULyraEquipmentInstance {
        -Instigator : UObject*
        -SpawnedActors : TArray~AActor~
        +OnEquipped()
        +OnUnequipped()
        +SpawnEquipmentActors()
        +DestroyEquipmentActors()
        +GetPawn() APawn
    }
    
    class ULyraWeaponInstance {
        <<Specialized Equipment>>
        +EquippedAnimSet
        +UnequippedAnimSet
    }
    
    class ULyraAbilitySet {
        <<DataAsset>>
        +GrantedGameplayAbilities
        +GrantedGameplayEffects
        +GrantedAttributes
        +GiveToAbilitySystem()
    }
    
    class ULyraQuickBarComponent {
        -Slots : TArray~ULyraInventoryItemInstance~
        -ActiveSlotIndex : int32
        -EquippedItem : ULyraEquipmentInstance*
        +SetActiveSlotIndex()
        +EquipItemInSlot()
        +UnequipItemInSlot()
    }
    
    %% Inheritance
    ULyraEquipmentInstance <|-- ULyraWeaponInstance : extends
    
    %% Composition (owns)
    ULyraEquipmentManagerComponent *-- "1" FLyraEquipmentList : contains
    FLyraEquipmentList *-- "*" FLyraAppliedEquipmentEntry : contains
    FLyraAppliedEquipmentEntry *-- "1" ULyraEquipmentInstance : has
    
    %% Association (uses/references)
    ULyraEquipmentDefinition ..> ULyraEquipmentInstance : creates
    ULyraEquipmentDefinition ..> ULyraAbilitySet : grants
    ULyraQuickBarComponent ..> ULyraEquipmentManagerComponent : uses
    ULyraQuickBarComponent ..> ULyraEquipmentInstance : equips
